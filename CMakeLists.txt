cmake_minimum_required(VERSION 3.16)
project(TexasHoldemOddsEngine VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

option(HOLDEM_NATIVE "Optimize for the build machine's CPU (-march=native)" ON)
option(HOLDEM_BUILD_TESTS "Build tests" ON)

find_package(Threads REQUIRED)

if(MSVC)
  # The evaluator's lookup tables are built at compile time.
  add_compile_options(/W4 /permissive- /constexpr:steps10000000)
  if(HOLDEM_NATIVE)
    add_compile_options(/arch:AVX2)
  endif()
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
  if(HOLDEM_NATIVE)
    add_compile_options(-march=native)
  else()
    # popcnt is on every x86-64 CPU since ~2008; without it popcount is a library call.
    include(CheckCXXCompilerFlag)
    check_cxx_compiler_flag(-mpopcnt HOLDEM_HAS_MPOPCNT)
    if(HOLDEM_HAS_MPOPCNT)
      add_compile_options(-mpopcnt)
    endif()
  endif()
endif()

# Evaluator, simulator, card parsing, starting-hand indexing.
add_library(holdem_core STATIC
  src/cards.cpp
  src/evaluator.cpp
  src/simulator.cpp
  src/starting_hands.cpp
)
target_include_directories(holdem_core PUBLIC include)
target_link_libraries(holdem_core PUBLIC Threads::Threads)

# Precomputed 169-hand odds table (src/preflop_table.inc, generated by `precompute`).
add_library(holdem STATIC src/preflop.cpp)
target_link_libraries(holdem PUBLIC holdem_core)

add_executable(holdem_odds apps/main.cpp)
target_link_libraries(holdem_odds PRIVATE holdem)

add_executable(precompute tools/precompute.cpp)
target_link_libraries(precompute PRIVATE holdem_core)

add_executable(benchmark bench/benchmark.cpp)
target_link_libraries(benchmark PRIVATE holdem)

if(HOLDEM_BUILD_TESTS)
  enable_testing()
  add_executable(test_main tests/test_main.cpp)
  target_link_libraries(test_main PRIVATE holdem)
  add_test(NAME unit COMMAND test_main)

  add_executable(test_exhaustive tests/test_exhaustive.cpp)
  target_link_libraries(test_exhaustive PRIVATE holdem_core)
  add_test(NAME exhaustive_7card COMMAND test_exhaustive)
endif()
